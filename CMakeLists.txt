cmake_minimum_required(VERSION 3.26)
project(audio_repeater LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Static C/C++ runtime on Windows, including the dependency.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
message(STATUS "Compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
if(MSVC)
  message(STATUS "Windows SDK: ${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}; toolset: ${CMAKE_VS_PLATFORM_TOOLSET}; architecture: ${CMAKE_VS_PLATFORM_NAME}")
endif()
include(CTest)
add_subdirectory(third_party)
add_library(audio_core STATIC src/audio_core.cpp)
target_include_directories(audio_core PUBLIC src)
target_link_libraries(audio_core PRIVATE SampleRate::samplerate)
function(strict_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endfunction()
strict_warnings(audio_core)
if(BUILD_TESTING)
  # Instrument C allocations in every vendor translation unit as well as C++ new.
  # The production library contains no instrumentation when BUILD_TESTING=OFF.
  if(MSVC)
    target_compile_options(samplerate PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/tests/allocator_hooks.h)
  else()
    target_compile_options(samplerate PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/tests/allocator_hooks.h)
  endif()
  add_executable(audio_core_tests tests/audio_core_tests.cpp tests/allocator_hooks.cpp)
  target_link_libraries(audio_core_tests PRIVATE audio_core SampleRate::samplerate)
  strict_warnings(audio_core_tests)
  foreach(suite formats fifo src quality drift allocation)
    add_test(NAME core_${suite} COMMAND audio_core_tests ${suite})
    set_tests_properties(core_${suite} PROPERTIES TIMEOUT 240)
  endforeach()
  # Compile actual source mutations in the build tree. Each must fail the same
  # behavior test that the unmodified core passes; production has no mutation API.
  file(READ src/audio_core.cpp original_core)
  foreach(mutation sign capacity)
    if(mutation STREQUAL "sign")
      string(REPLACE "return ppm();" "return -ppm();" mutated_core "${original_core}")
    else()
      string(REPLACE "queued_input() + static_cast<double>(frames) > static_cast<double>(capacity())"
        "false" mutated_core "${original_core}")
    endif()
    if(mutated_core STREQUAL original_core)
      message(FATAL_ERROR "Mutation did not alter source")
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mutated_${mutation}.cpp" "${mutated_core}")
    add_executable(mutated_${mutation} "${CMAKE_CURRENT_BINARY_DIR}/mutated_${mutation}.cpp"
      tests/audio_core_tests.cpp tests/allocator_hooks.cpp)
    target_include_directories(mutated_${mutation} PRIVATE src)
    target_link_libraries(mutated_${mutation} PRIVATE SampleRate::samplerate)
    strict_warnings(mutated_${mutation})
  endforeach()
  add_test(NAME mutation_control_sign COMMAND audio_core_tests mutation-sign)
  add_test(NAME mutation_control_capacity COMMAND audio_core_tests mutation-capacity)
  add_test(NAME rejects_reverse_feedback COMMAND mutated_sign mutation-sign)
  add_test(NAME rejects_capacity_violation COMMAND mutated_capacity mutation-capacity)
  set_tests_properties(rejects_reverse_feedback rejects_capacity_violation
    PROPERTIES WILL_FAIL TRUE TIMEOUT 30)
endif()
